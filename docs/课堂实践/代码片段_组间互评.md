 594  post('/api/projects/:id/assign-reviews', (ctx) => {
 595    const p = store.find('projects', ctx.params.id);
 596    need(p, 404, '项目不存在');
 597    requireMembership(ctx, p.courseId, ['teacher', 'admin']);
 598    const teams = store.where('teams', (t) => t.projectId === p.id);
 599    need(teams.length >= 2, 400, '至少需要 2 个小组才能分配互评');
 600    const per = Math.min(Number(ctx.body.reviewersPerSubmission) || 3, teams.length - 1);
 601    let created = 0;
 602    for (const s of store.where('submissions', (x) => x.projectId === p.id)) {
 603      for (const t of teams.filter((x) => x.id !== s.teamId).slice(0, per)) {
 604        if (store.first('assignments', (a) => a.submissionId === s.id && a.reviewerTeamId === t.id)) continue;
 605        store.insert('assignments', {
 606          projectId: p.id, submissionId: s.id, reviewerTeamId: t.id,
 607          assignedById: ctx.user.id, assignedAt: now(), status: 'pending',
 608          dueDate: ctx.body.dueDate || new Date(Date.now() + 5 * dayMs).toISOString()
 609        });
 610        created++;
 611      }
 612    }
 613    act.log({ courseId: p.courseId, projectId: p.id, actorId: ctx.user.id, action: 'review_assign', detail: created + ' 个任务' });
 614    store.save();
 615    return { created, reviewersPerSubmission: per, teams: teams.length };
 616  });
